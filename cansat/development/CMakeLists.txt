cmake_minimum_required(VERSION 3.16)
project(ino_module_scaffolds LANGUAGES CXX)
enable_testing()
foreach(module IN ITEMS main camera drive)
  add_executable(${module}_scaffold tests/${module}_smoke.cpp)
  target_include_directories(${module}_scaffold PRIVATE include)
  target_compile_features(${module}_scaffold PRIVATE cxx_std_11)
  add_test(NAME ${module}_scaffold COMMAND ${module}_scaffold)
endforeach()
